cmake_minimum_required(VERSION 3.20)
project(nsx_hardware_startup C CXX ASM)
include("${NSX_TEST_SDK_ROOT}/tests/hardware/common/firmware.cmake")
nsx_add_test_firmware(startup_test main.cpp)

if(NSX_TEST_FAULT STREQUAL "assertion")
    target_compile_definitions(startup_test PRIVATE NSX_TEST_FAIL_ASSERTION)
elseif(NSX_TEST_FAULT STREQUAL "timeout")
    target_compile_definitions(startup_test PRIVATE NSX_TEST_NO_COMPLETION)
elseif(NSX_TEST_FAULT MATCHES "^omit-(data|bss|constructors)$")
    # The builder supplies an artifact-only modified startup source.
    if(NOT NSX_TEST_STARTUP_OVERRIDE)
        message(FATAL_ERROR "Omission requires a startup override")
    endif()
elseif(NOT NSX_TEST_FAULT STREQUAL "none")
    message(FATAL_ERROR "Unknown fault-injection mode")
endif()
