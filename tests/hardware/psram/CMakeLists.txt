cmake_minimum_required(VERSION 3.20)
project(nsx_psram_blocking C CXX ASM)
include("${NSX_TEST_HARNESS_ROOT}/tests/hardware/common/firmware.cmake")
nsx_add_test_firmware(psram_blocking main.c MODULES interrupt psram)
if(NSX_TEST_PSRAM_DEVICE_OVERRIDE)
    get_target_property(psram_sources nsx_psram SOURCES)
    list(FILTER psram_sources EXCLUDE REGEX "am_devices_mspi_psram_aps25616ba_1p2v\\.c$")
    set_property(TARGET nsx_psram PROPERTY SOURCES ${psram_sources} "${NSX_TEST_PSRAM_DEVICE_OVERRIDE}")
    target_include_directories(nsx_psram PRIVATE "${NSX_TEST_SDK_ROOT}/modules/nsx-psram/src/apollo5/devices")
endif()
target_link_options(psram_blocking PRIVATE "LINKER:--wrap=am_hal_cachectrl_dcache_invalidate")
if(NSX_TEST_FAULT STREQUAL "psram-transfer-flush")
    target_compile_definitions(psram_blocking PRIVATE NSX_TEST_TRANSFER_FLUSH)
    target_link_options(psram_blocking PRIVATE "LINKER:--wrap=nsx_psram_write")
endif()
