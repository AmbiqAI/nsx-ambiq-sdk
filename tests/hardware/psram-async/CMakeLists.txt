cmake_minimum_required(VERSION 3.20)
project(nsx_psram_async C CXX ASM)
include("${NSX_TEST_HARNESS_ROOT}/tests/hardware/common/firmware.cmake")
nsx_add_test_firmware(psram_async main.c MODULES interrupt psram)
set(NSX_TEST_PSRAM_CLOCK 48000000 CACHE STRING "Requested HAL PSRAM clock setting")
option(NSX_TEST_CPU_HIGH "Use the supported high CPU mode" OFF)
if(NOT NSX_TEST_PSRAM_CLOCK MATCHES "^(48000000|96000000)$")
    message(FATAL_ERROR "Async qualification supports HAL clock settings 48/96 MHz")
endif()
target_compile_definitions(psram_async PRIVATE
    NSX_TEST_PSRAM_CLOCK=${NSX_TEST_PSRAM_CLOCK}
    NSX_TEST_CPU_HIGH=$<BOOL:${NSX_TEST_CPU_HIGH}>)
target_compile_definitions(psram_async PRIVATE NSX_TEST_ASYNC_DEPTH=${NSX_PSRAM_ASYNC_DEPTH})
if(NSX_TEST_FAULT STREQUAL "async-done-early")
    target_compile_definitions(psram_async PRIVATE NSX_TEST_DONE_EARLY)
    target_link_options(psram_async PRIVATE "LINKER:--wrap=nsx_psram_read_async")
elseif(NSX_TEST_FAULT STREQUAL "async-global-drain")
    target_compile_definitions(psram_async PRIVATE NSX_TEST_GLOBAL_DRAIN)
    target_link_options(psram_async PRIVATE "LINKER:--wrap=nsx_psram_client_drain")
elseif(NSX_TEST_FAULT STREQUAL "async-drain-early")
    target_compile_definitions(psram_async PRIVATE NSX_TEST_DRAIN_EARLY)
    target_link_options(psram_async PRIVATE "LINKER:--wrap=nsx_psram_client_drain")
endif()
