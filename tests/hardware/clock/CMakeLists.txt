cmake_minimum_required(VERSION 3.20)
project(nsx_hardware_clock C CXX ASM)
include("${NSX_TEST_HARNESS_ROOT}/tests/hardware/common/firmware.cmake")
nsx_add_test_firmware(clock_test main.c MODULES power)
if(NSX_TEST_FAULT STREQUAL "clock-stale-publication")
    target_compile_definitions(clock_test PRIVATE CLOCK_STALE_PUBLICATION)
elseif(NSX_TEST_FAULT STREQUAL "clock-no-reload")
    target_compile_definitions(clock_test PRIVATE CLOCK_NO_RELOAD)
elseif(NSX_TEST_FAULT STREQUAL "clock-wrong-scaling")
    target_compile_definitions(clock_test PRIVATE CLOCK_WRONG_SCALING)
elseif(NOT NSX_TEST_FAULT STREQUAL "none")
    message(FATAL_ERROR "Unknown clock diagnostic fault")
endif()
